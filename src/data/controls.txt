# Управление мини-плеером Bad Apple.
# Формат: клавиша действие # что делает.
# Строки с # -- комментарии. Плеер читает этот файл при старте.
# Клавиши: space, left, right, up, down, home, end, enter, escape, буква.

space play_pause     # play / pause: пуск и пауза
left seek_back       # перемотка назад на 10 секунд
right seek_forward   # перемотка вперёд на 10 секунд
up volume_up         # громкость + (звук громче)
down volume_down     # громкость - (звук тише)
m mute               # выключить / включить звук (mute)
s stop               # стоп: в начало и на паузу
home restart         # вернуться в самое начало видео
[ size_down          # размер видео меньше (предыдущий размер)
] size_up            # размер видео больше (следующий размер)
q quit               # выйти из плеера обратно в шелл
a audio_quality      # следующее качество звука: 4k -> 8k -> 11k -> 16k -> 32k
t track            # переключить трек: Bad Apple / Rickroll
